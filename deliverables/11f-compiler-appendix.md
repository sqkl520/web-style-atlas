```powershell
        if ($lines[$i] -match '^### 方案 (\d+)') { $starts += [pscustomobject]@{ idx = $i; n = [int]$Matches[1] } }
    }
    if ($starts.Count -ne 10) { throw "$f 期望 10 个方案块，实际 $($starts.Count) —— 立即停止，未做任何写入" }

    # 从后往前插入，避免索引漂移
    $done = 0
    for ($k = $starts.Count - 1; $k -ge 0; $k--) {
        $s = $starts[$k]
        $endEx = $lines.Count
        for ($j = $s.idx + 1; $j -lt $lines.Count; $j++) {
            if ($lines[$j] -match '^### 方案 ' -or $lines[$j] -match '^#{1,2} ') { $endEx = $j; break }
        }
        # 回退到该案最后一个内容行（跳过尾部空行与 --- 分隔线）
        $last = $endEx - 1
        while ($last -gt $s.idx -and ($lines[$last].Trim() -eq '' -or $lines[$last] -match '^---\s*$')) { $last-- }
        if ($last -le $s.idx) { throw "$f 方案 $($s.n) 未找到内容行 —— 立即停止" }

        $rec = $D[$s.n]
        if (-not $rec) { throw "$f 方案 $($s.n) 缺少数据 —— 立即停止" }
        $anno = @(
            "版本：$($rec.ver)",
            "校验：正文对比度 $($rec.c1):1（07 号实测，AA 与 AAA 通过）；字体授权$($rec.f)",
            "审查：反AI味 $($rec.s)/10｜差异化：$($rec.d)｜红队风险：$($rec.q)"
        )
        $lines.InsertRange($last + 1, [string[]]$anno)
        $done++
    }
    $grandInserted += $done
    [System.IO.File]::WriteAllLines($src, $lines, (New-Object System.Text.UTF8Encoding($false)))
    $report.Add("$f : 备份=$bak\$f  插入案数=$done  插入行数=$($done*3)")
}

# ---------- 落盘后断言 ----------
$totalV = 0; $totalJ = 0; $totalS = 0
foreach ($f in $files) {
    $t = [System.IO.File]::ReadAllText((Join-Path $dir $f))
    $v = ([regex]::Matches($t, '(?m)^版本：')).Count
    $j = ([regex]::Matches($t, '(?m)^校验：')).Count
    $s = ([regex]::Matches($t, '(?m)^审查：')).Count
    $totalV += $v; $totalJ += $j; $totalS += $s
    $report.Add("$f : 版本：=$v  校验：=$j  审查：=$s")
}
$report.Add("合计：插入案数=$grandInserted（期望 30）  版本：=$totalV  校验：=$totalJ  审查：=$totalS（各期望 30）")
if ($grandInserted -ne 30 -or $totalV -ne 30 -or $totalJ -ne 30 -or $totalS -ne 30) { throw "断言失败：三行未按 30 案齐全插入" }
$report.Add("断言通过：30 案各成功插入 3 行，合计 90 行。备份目录=$bak")
$report | ForEach-Object { $_ }
```

### 11f-7.6 为什么对比度列标注了"07 号实测"

`07` 号的 C1 是在**它当时看到的背景值**上算出的。此后有 5 个方案的背景按 `08` 号第 14 节的表调整过（方案 19、20、21、23、25），另有方案 6 的背景与 4 号共享同一次改动。这几次调整的方向是"浅底更浅"（20）或"深字底更暗"（19、21、23、25、6），**只会抬高正文与背景的对比度，不会降低**，所以 `07` 号的实测值在这几案上是**保守下界**，可以直接沿用而不失真。这也是三行标注里写成"07 号实测"而不是笼统写"已达标"的原因：数值有出处，读者可以自己回查。
